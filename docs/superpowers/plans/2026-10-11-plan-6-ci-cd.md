type Statement = {
  Action: string | string[];
  Resource?: unknown;
  Condition?: Record<string, Record<string, string>>;
};

const roleNamed = (template: Template, name: string) => {
  const roles = Object.values(template.findResources("AWS::IAM::Role")).filter(
    (r) => r.Properties.RoleName === name,
  );
  expect(roles).toHaveLength(1);
  return roles[0] as { Properties: { AssumeRolePolicyDocument: { Statement: Statement[] } } };

    .filter(([id]) => id.startsWith(roleLogicalIdPrefix))
    .flatMap(([, p]) => p.Properties.PolicyDocument.Statement as Statement[]);

`infra/lib/ci-stack.ts`:

```ts
import { Duration, Stack, type StackProps } from "aws-cdk-lib";
import * as iam from "aws-cdk-lib/aws-iam";
import type { Construct } from "constructs";

const GITHUB_ISSUER = "token.actions.githubusercontent.com";
/** Default CDK bootstrap qualifier. */
const QUALIFIER = "hnb659fds";

/**
 * GitHub Actions → AWS without stored keys. Deployed by hand once (`cdk deploy BlogCi`);
 * the deploy workflow never touches this stack, so it cannot widen its own permissions.
 */
export class CiStack extends Stack {
  readonly deployRole: iam.Role;
  readonly gcRole: iam.Role;

  constructor(scope: Construct, id: string, props: StackProps & { githubRepo: string }) {
    super(scope, id, props);
    const { account, region } = this;

    const provider = new iam.OidcProviderNative(this, "GitHubOidc", {
      url: `https://${GITHUB_ISSUER}`,
      clientIds: ["sts.amazonaws.com"],
    });
    // Only workflows running on main of this repository (not forks, not pull requests).
    const githubMain = () =>
      new iam.WebIdentityPrincipal(provider.oidcProviderArn, {
        StringEquals: {
          [`${GITHUB_ISSUER}:aud`]: "sts.amazonaws.com",
          [`${GITHUB_ISSUER}:sub`]: `repo:${props.githubRepo}:ref:refs/heads/main`,
        },
      });

    this.deployRole = new iam.Role(this, "DeployRole", {
      roleName: "blog-github-deploy",
      description: "GitHub Actions (main): cdk deploy through the bootstrap roles, then rebuild the site",
      assumedBy: githubMain(),
      maxSessionDuration: Duration.hours(1),
    });
    // Every resource change goes through the bootstrap roles (deploy, file/image publishing, lookup).
    this.deployRole.addToPolicy(
      new iam.PolicyStatement({
        actions: ["sts:AssumeRole"],
        resources: [`arn:aws:iam::${account}:role/cdk-${QUALIFIER}-*-${account}-*`],
      }),
    );
    this.deployRole.addToPolicy(
      new iam.PolicyStatement({
        actions: ["sqs:SendMessage"],
        resources: [`arn:aws:sqs:${region}:${account}:Blog-BuildBuildQueue*`],
      }),
    );
    this.deployRole.addToPolicy(
      new iam.PolicyStatement({
        actions: ["dynamodb:GetItem"],
        resources: [`arn:aws:dynamodb:${region}:${account}:table/BlogData-BlogTable*`],
      }),
    );

    this.gcRole = new iam.Role(this, "GcRole", {
      roleName: "blog-github-gc",
      description: "GitHub Actions (main): cdk gc of unused bootstrap assets",
      assumedBy: githubMain(),
      maxSessionDuration: Duration.hours(1),
    });
    // cdk gc reads every template to find assets still in use.
    this.gcRole.addToPolicy(
      new iam.PolicyStatement({
        actions: [
          "cloudformation:ListStacks",
          "cloudformation:DescribeStacks",
          "cloudformation:GetTemplate",
        ],
        resources: ["*"],
      }),
    );
    this.gcRole.addToPolicy(
      new iam.PolicyStatement({
        actions: ["ssm:GetParameter"],
        resources: [`arn:aws:ssm:*:${account}:parameter/cdk-bootstrap/${QUALIFIER}/version`],
      }),
    );
    const assetBuckets = `arn:aws:s3:::cdk-${QUALIFIER}-assets-${account}-*`;
    this.gcRole.addToPolicy(
      new iam.PolicyStatement({
        actions: ["s3:ListBucket", "s3:GetBucketLocation"],
        resources: [assetBuckets],
      }),
    );
    this.gcRole.addToPolicy(
      new iam.PolicyStatement({
        actions: ["s3:GetObjectTagging", "s3:PutObjectTagging", "s3:DeleteObject"],

```bash
#!/usr/bin/env bash
# Rebuild the site with the image just deployed, then wait until BUILD/LATEST reports the result
# of *this* build (updatedAt newer than the moment the message was sent). Spec 5.5: 5 minutes.
set -euo pipefail

OUTPUTS=${1:-infra/cdk-outputs.json}
REGION=${AWS_REGION:-ap-northeast-1}
QUEUE_URL=$(jq -r '.Blog.BuildQueueUrl' "$OUTPUTS")
TABLE=$(jq -r '.BlogData | to_entries[] | select(.key | startswith("ExportsOutputRefBlogTable")) | .value' "$OUTPUTS")
if [[ -z "$QUEUE_URL" || "$QUEUE_URL" == "null" || -z "$TABLE" ]]; then
  echo "cannot read the build queue / table from $OUTPUTS" >&2
  exit 1
fi

SENT_AT=$(date -u +%Y-%m-%dT%H:%M:%S.000Z)
aws sqs send-message --region "$REGION" --queue-url "$QUEUE_URL" \
  --message-body '{"reason":"deploy"}' --message-group-id site \
  --message-deduplication-id "deploy-${GITHUB_RUN_ID:-local}-$(date +%s)" >/dev/null
echo "build requested at $SENT_AT"

for _ in $(seq 1 60); do
  read -r STATUS UPDATED < <(aws dynamodb get-item --region "$REGION" --table-name "$TABLE" \
    --key '{"PK":{"S":"BUILD"},"SK":{"S":"LATEST"}}' \
    --query '[Item.status.S, Item.updatedAt.S]' --output text)
  if [[ "$UPDATED" > "$SENT_AT" ]]; then
    case "$STATUS" in
      success) echo "build success ($UPDATED)"; exit 0 ;;
      failed) echo "build failed ($UPDATED) — see /aws/lambda/blog-build" >&2; exit 1 ;;
    esac
  fi
  sleep 5
done
echo "the build did not finish within 5 minutes" >&2
exit 1
```

# oki-dev blog

Personal serverless blog: **https://blog.oki-dev.xyz** (Vietnamese / Japanese).

- `apps/site` — Astro static site (built by a Lambda after each publish)
- `apps/admin` — React admin at `admin.oki-dev.xyz` (Cognito login, Markdown editor)
- `services/api` — HTTP API (Lambda, DynamoDB), `services/build` — site build Lambda (container)
- `packages/*` — shared types, Markdown pipeline, design tokens
- `infra` — AWS CDK (Tokyo + us-east-1 certificate), `e2e` — read-only checks against production

## Develop

```bash
pnpm install
pnpm test        # every package
pnpm lint && pnpm typecheck
pnpm --filter @blog/site dev
pnpm --filter @blog/admin dev:config && pnpm --filter @blog/admin dev
```

## Ship

`main` only accepts pull requests (squash, CI must pass). Merging deploys to production through GitHub Actions (OIDC, no stored keys), rebuilds the site and runs the read-only E2E checks. See `docs/runbooks/first-deploy.md`.
```

```markdown
## CI/CD (GitHub Actions)

- Repo: `sonnguyendev157/my-blog` (public). `main`: chỉ nhận PR, squash, check `ci` phải xanh, không bypass.
- Quy trình: `git checkout -b feat/…` → push → `gh pr create` → CI xanh → `gh pr merge --squash` → workflow `deploy` tự chạy (deploy 3 stack, build lại site, smoke + E2E chỉ đọc).
- Quyền AWS: role `blog-github-deploy` / `blog-github-gc` qua OIDC (stack `BlogCi`, deploy tay: `pnpm --filter @blog/infra exec cdk deploy BlogCi`). Không có access key/GitHub Secret.
- Deploy lại không đổi code: `gh workflow run deploy.yml`.
- Dọn asset CDK: workflow `gc` chạy ngày 1 hằng tháng; xem trước bằng `gh workflow run gc.yml -f action=print`.
- E2E local (chỉ đọc prod): `pnpm --filter @blog/e2e e2e`.
- Workflow đỏ → GitHub gửi email; xem `gh run view --log-failed`. Không tự rollback: sửa qua PR mới.
```
